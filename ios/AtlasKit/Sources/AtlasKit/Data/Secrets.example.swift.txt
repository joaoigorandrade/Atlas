import Foundation

/// The credentials the app calls out with, in the source, on purpose: this is
/// the development phase and the app talks to Supabase and OpenRouter itself
/// rather than through the web app.
///
/// **This file is not committed** (`.gitignore`) — `Secrets.example.swift` is
/// the committed template. The repo is public, and an OpenRouter key is a spend
/// credential: pushed once, it is scraped within minutes and every call until
/// the revoke is billed to this account. Copy the example, fill it in, build.
///
/// ponytail: two constants and no configuration layer. The refactor when this
/// stops being the development phase is one move — the OpenRouter calls go back
/// behind `/api/generate` (which still exists and is still deployed) and this
/// file is deleted, key and all.
// Copy to `Secrets.swift` (same directory) and fill in from the web app's
// `.env.local`, which already holds all four values.
public enum Secrets {
    /// Supabase, for auth. The publishable key is designed for clients and RLS
    /// is the real access control, so this one is a URL and a public identifier
    /// — it is here for convenience, not because it is dangerous.
    public static let supabaseURL = URL(string: "https://YOUR-PROJECT.supabase.co")!
    public static let supabasePublishableKey = "sb_publishable_…"

    /// OpenRouter, for content. This one *is* dangerous: it spends money and it
    /// ships inside the binary, where `strings` on an extracted .ipa finds it.
    /// Fine while the only build is on this machine; not fine the moment one
    /// leaves it.
    public static let openRouterKey = "sk-or-v1-…"
    public static let openRouterModel = "google/gemini-2.5-flash"
}
